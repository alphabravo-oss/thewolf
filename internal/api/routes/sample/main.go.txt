// Intentionally vulnerable training sample. Never deploy this application.
package main

import (
	"crypto/md5"
	"database/sql"
	"fmt"
	"net/http"
	"os/exec"
)

func run(w http.ResponseWriter, r *http.Request) {
	output, err := exec.Command("sh", "-c", r.URL.Query().Get("command")).Output()
	if err != nil {
		http.Error(w, "command failed", http.StatusInternalServerError)
		return
	}
	_, _ = w.Write(output)
}

func lookup(db *sql.DB, name string) (*sql.Rows, error) {
	query := fmt.Sprintf("SELECT * FROM users WHERE name = '%s'", name)
	return db.Query(query)
}

func weakHash(password string) string {
	return fmt.Sprintf("%x", md5.Sum([]byte(password)))
}

func main() {
	// Deliberately inert: scanners inspect the vulnerable handlers above.
	fmt.Println("Wolf training sample: intentionally vulnerable; do not deploy.")
}
